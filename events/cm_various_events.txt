character_event = {
	id = CM.2000
	desc = EVTDESC_CM_2000
	picture = GFX_evt_emissary

	ai = no
	capable_only = yes
	prisoner = no
	only_playable = yes
	min_age = 16
	has_dlc = "Charlemagne"
	
	trigger = {	
		higher_tier_than = DUKE
	}
	
	mean_time_to_happen = {
		years = 1000
	}
	
	option = {
		name = EVTOPTA_CM_2000
		capital_scope = {
			ROOT = {
				if = {
					limit = {
						PREV = {
							female_warrior_culture_trigger = yes
						}
					}
					create_character = {
						random_traits = no
						dynasty = random
						religion = ROOT
						culture = PREV
						age = 35
						trait = brave
						trait = strong
						trait = skilled_tactician
						trait = master_warrior
						female = yes
						historical = yes
						attributes = {
							martial = 6
						}
					}
				}
				else = {
					create_character = {
						random_traits = no
						dynasty = random
						religion = ROOT
						culture = PREV
						age = 35
						trait = brave
						trait = strong
						trait = skilled_tactician
						trait = master_warrior
						historical = yes
						attributes = {
							martial = 6
						}
					}
				}				
				new_character = {
					set_character_flag = hero_knight
					opinion = {
						modifier = opinion_loyal_servant
						who = ROOT
						years = 100
					}
					if = {
						limit = { religion = the_seven }
						knight_character_effect = yes
					}
					ROOT = {
						if = {
							limit = { ai = no }
							chronicle = {
								entry = CHRONICLE_HERO_KNIGHT_APPEARS
								portrait = [Prev.GetID]
							}
						}
					}
				}
			}	
		}	
	}
	option = {
		name = EVTOPTB_CM_2000
		prestige = 10
	}
}

character_event = {
	id = CM.2020
	
	hide_window = yes

	is_triggered_only = yes
	ai = no
	min_age = 16
	has_dlc = "Charlemagne"	
	
	trigger = {		
		any_child = {
			age = 9
			NOT = { trait = incapable }
			educator = {
				has_character_flag = hero_knight
				NOT = { has_character_flag = knight_adventure }
				prisoner = no
			}
		}
	}
	
	immediate = {
		random_child = {
			limit = {
				age = 9
				NOT = { trait = incapable }
				OR = {
					is_female = no
					female_get_masculine_event_trigger = yes
				}
				educator = {
					has_character_flag = hero_knight
					NOT = { has_character_flag = knight_adventure }
					prisoner = no
				}
			}
			character_event = { id = CM.2021 }
		}
	}
}

character_event = {
	id = CM.2021

	hide_window = yes

	is_triggered_only = yes

	immediate = {
		random_list = {
			10 = { FROM = { character_event = { id = CM.2022 } } } # Brave
			10 = { FROM = { character_event = { id = CM.2024 } } } # Strong
			10 = { FROM = { character_event = { id = CM.2026 } } } # Duelist
			10 = { FROM = { character_event = { id = CM.2028 } } } # Just
			1 = { character_event = { id = CM.2029 } } # Wounded or maimed
		}
	}
}

# Child learns bravery
character_event = {
	id = CM.2022
	desc = EVTDESC_CM_2022
	picture = GFX_evt_child_sword

	is_triggered_only = yes

	trigger = {
		FROM = { NOT = { trait = brave } }
		FROM = { educator = { NOT = { trait = craven } } }
	}
	
	option = {
		name = EVTOPTA_CM_2022
		trigger = {
			NOT = { trait = craven }
		}
		FROM = {
			add_trait = brave
		}
	}
	option = {
		name = EVTOPTA_CM_2022
		trigger = {
			trait = craven
		}
		FROM = {
			add_trait = brave
		}
	}
}

# Child is wounded or maimed while practising with knight
character_event = {
	id = CM.2029

	hide_window = yes
	
	is_triggered_only = yes
	
	trigger = {
		NOT = { trait = wounded }
		NOT = { has_severe_disability_trigger = yes }
	}

	immediate = {
		random_list = {
			80 = { add_trait_silently_wounded_effect = yes }
			20 = { add_maimed_trait_effect = yes }
		}
		FROMFROM = { character_event = { id = CM.2030 } }
	}
}

character_event = {
	id = CM.2030
	desc = EVTDESC_CM_2030
	picture = GFX_evt_child_sword

	is_triggered_only = yes
